基础查询

SELECT 列名称 FROM 表名称
SELECT DISTINCT 列名称 FROM 表名称
SELECT 列名称 FROM 表名称 WHERE 列 运算符 值
INSERT INTO 表名称 VALUES (值1, 值2,....)
UPDATE 表名称 SET 列名称 = 新值 WHERE 列名称 = 某值
DELETE FROM 表名称 WHERE 列名称 = 值
SELECT TOP 列数额 * FROM 表名称
SELECT TOP 百分比 PERCENT * FROM 表名称

条件筛选

SELECT * FROM 表名称 WHERE 列名称 IN ('列内容1','列内容2')
SELECT * FROM 表名称 WHERE 列名称 BETWEEN '列内容1' AND '列内容2'  -- 输出为列内容1到列内容2区间
SELECT * FROM 表名称 WHERE 列名称 NOT BETWEEN '列内容1' AND '列内容2'  -- 输出为列内容1到列内容2以外的值
  • AND 和 OR 可在 WHERE 子语句中把两个或多个条件结合起来。
  • 如果第一个条件和第二个条件都成立，则 AND 运算符显示一条记录。
  • 如果第一个条件和第二个条件中只要有一个成立，则 OR 运算符显示一条记录。

别名与连接

SELECT 列名称1 AS 别名1, 列名称2 AS 别名2 FROM 表名称

SELECT 表名称1.列名称1, 表名称1.列名称2, 表名称2.列名称1
FROM 表名称1, 表名称2
WHERE 表名称1 = 表名称2

SELECT Persons.LastName, Persons.FirstName, Orders.OrderNo
FROM Persons
INNER JOIN Orders ON Persons.Id_P = Orders.Id_P
ORDER BY Persons.LastName

SELECT 表名称1.列名称, 表名称2.列名称
FROM 表名称1
CROSS JOIN 表名称2  -- 完全连接 (FULL OUTER JOIN)

排序与分组

-- ORDER BY 语句用于根据指定的列对结果集进行排序
-- ORDER BY 语句默认按照升序对记录进行排序
-- 如果希望按照降序排序，可以使用 DESC 关键字

SELECT 列名称 FROM 表名称
GROUP BY 列名称
HAVING COUNT(*) > 1  -- 查询列重复值 > 1 的值

高级查询

-- 子查询
SELECT 列名称 FROM 表名称 WHERE 列名称 = (SELECT 列名称 FROM 另一表 WHERE 条件);

-- EXISTS 子查询
SELECT 列名称 FROM 表名称 WHERE EXISTS (SELECT 1 FROM 另一表 WHERE 条件);

-- ANY 和 ALL
SELECT 列名称 FROM 表名称 WHERE 列名称 > ANY (SELECT 列名称 FROM 另一表 WHERE 条件);
SELECT 列名称 FROM 表名称 WHERE 列名称 > ALL (SELECT 列名称 FROM 另一表 WHERE 条件);

联合查询

-- UNION：合并并去重
SELECT 列名称 FROM 表名称1
UNION
SELECT 列名称 FROM 表名称2;

-- UNION ALL：合并所有结果，不去重
SELECT 列名称 FROM 表名称1
UNION ALL
SELECT 列名称 FROM 表名称2;

窗口函数

-- ROW_NUMBER
SELECT 列名称, ROW_NUMBER() OVER (ORDER BY 列名称) AS 行号 FROM 表名称;

-- RANK 和 DENSE_RANK
SELECT 列名称, RANK() OVER (ORDER BY 列名称) AS 排名 FROM 表名称;
SELECT 列名称, DENSE_RANK() OVER (ORDER BY 列名称) AS 密集排名 FROM 表名称;

-- NTILE
SELECT 列名称, NTILE(4) OVER (ORDER BY 列名称) AS 四分位数 FROM 表名称;

-- 窗口聚合函数 (SUM, AVG, MIN, MAX)
SELECT 列名称,
    SUM(列名称) OVER (PARTITION BY 列名称2) AS 分区总和,
    AVG(列名称) OVER (PARTITION BY 列名称2) AS 分区平均值
FROM 表名称;

临时表与表变量

-- 临时表
CREATE TABLE #临时表名称 (列名称 数据类型, ...);

INSERT INTO #临时表名称
SELECT 列名称 FROM 表名称 WHERE 条件;

SELECT 列名称 FROM #临时表名称;

DROP TABLE #临时表名称;

-- 表变量
DECLARE @表变量 TABLE (列名称 数据类型, ...);

INSERT INTO @表变量
SELECT 列名称 FROM 表名称 WHERE 条件;

SELECT 列名称 FROM @表变量;

通用表表达式 (CTE)

-- 使用 CTE
WITH CTE名称 AS (
    SELECT 列名称 FROM 表名称 WHERE 条件
)
SELECT 列名称 FROM CTE名称 WHERE 条件;

-- 递归 CTE
WITH 递归CTE AS (
    SELECT 列名称 FROM 表名称 WHERE 条件
    UNION ALL
    SELECT 列名称 FROM 表名称, 递归CTE WHERE 条件
)
SELECT 列名称 FROM 递归CTE;

聚合与分组

-- GROUP BY 子句
SELECT 列名称, COUNT(*), SUM(列名称2), AVG(列名称2)
FROM 表名称
GROUP BY 列名称;

-- ROLLUP
SELECT 列名称1, 列名称2, SUM(列名称3)
FROM 表名称
GROUP BY ROLLUP (列名称1, 列名称2);

-- CUBE
SELECT 列名称1, 列名称2, SUM(列名称3)
FROM 表名称
GROUP BY CUBE (列名称1, 列名称2);
